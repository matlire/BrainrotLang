GENERAL ::= FUNCTION_DECL+ EOF

FUNCTION_DECL  ::= RET_TYPE IDENTIFIER "(" PARAMETER_LIST ")" COMPOUND_STATEMENT
PARAMETER      ::= TYPE IDENTIFIER
PARAMETER_LIST ::= PARAMETER ( "," PARAMETER )* | <none>

TYPE     ::= "int" | "float" | "ptr"
RET_TYPE ::= TYPE | "void"

STATEMENT ::=
      COMPOUND_STATEMENT
    | WHILE
    | FOR
    | IF
    | VAR_DECL ";"
    | ASSIGNMENT ";"
    | BREAK_STMT ";"
    | RETURN_STMT ";"
    | CALL_STMT ";"
    | PRINT_STMT ";"
    | IPRINT_STMT ";"
    | FPRINT_STMT ";"
    | EXPRESSION ";"

COMPOUND_STATEMENT ::= "{" STATEMENT* "}"

VAR_DECL   ::= TYPE IDENTIFIER [ "=" EXPRESSION ]
ASSIGNMENT ::= IDENTIFIER "=" EXPRESSION

BREAK_STMT  ::= "break"
RETURN_STMT ::= "return" [ EXPRESSION ]

CALL_STMT ::= "call" IDENTIFIER "(" ARGUMENT_LIST ")"

PRINT_STMT  ::= "print"  "(" EXPRESSION ")"
IPRINT_STMT ::= "iprint" "(" EXPRESSION ")"
FPRINT_STMT ::= "fprint" "(" EXPRESSION ")"

WHILE ::= "while" "(" EXPRESSION ")" STATEMENT

FOR      ::= "for" "(" FOR_INIT ";" FOR_COND ";" FOR_STEP ")" STATEMENT
FOR_INIT ::= VAR_DECL | ASSIGNMENT | <none>
FOR_COND ::= EXPRESSION | <none>
FOR_STEP ::= ASSIGNMENT | EXPRESSION | <none>

IF         ::= "if" "(" EXPRESSION ")" STATEMENT ELIF_CHAIN? ELSE?
ELIF_CHAIN ::= ELIF ELIF_CHAIN? | <none>
ELIF       ::= "elif" "(" EXPRESSION ")" STATEMENT
ELSE       ::= "else" STATEMENT

ARGUMENT_LIST ::= EXPRESSION ( "," EXPRESSION )* | <none>

EXPRESSION ::= OP_OR

OP_OR      ::= OP_AND    ( "||" OP_AND )*
OP_AND     ::= OP_EQ_NEQ ( "&&" OP_EQ_NEQ )*
OP_EQ_NEQ  ::= OP_CMP    ( ("==" | "!=") OP_CMP )*
OP_CMP     ::= OP_ADD    ( (">" | "<" | ">=" | "<=") OP_ADD )*
OP_ADD     ::= OP_MUL    ( ("+" | "-") OP_MUL )*
OP_MUL     ::= OP_POW    ( ("*" | "/") OP_POW )*
OP_POW     ::= OP_UNARY  ( "^" OP_POW )?
OP_UNARY   ::= ("!" | "-" | "+") OP_UNARY | PRIMARY

PRIMARY ::=
      BUILTIN_UNARY_CALL
    | DERIVATIVE_CALL
    | FUNCTION_CALL
    | IDENTIFIER
    | LITERAL
    | "(" EXPRESSION ")"

FUNCTION_CALL ::= IDENTIFIER "(" ARGUMENT_LIST ")"

BUILTIN_UNARY_CALL ::= BUILTIN_UNARY_NAME "(" EXPRESSION ")"
BUILTIN_UNARY_NAME ::=
      "floor"
    | "ceil"
    | "round"
    | "itof"
    | "ftoi"

DERIVATIVE_CALL ::= "d" "(" STRING_LITERAL "," IDENTIFIER "," NUMERIC_LITERAL ")"

LITERAL         ::= NUMERIC_LITERAL | STRING_LITERAL
NUMERIC_LITERAL ::= [0-9]+ ( "." [0-9]+ )?
STRING_LITERAL  ::= "\"" ( [^"\\\n] | "\\" . )* "\""
IDENTIFIER      ::= [a-zA-Z_][0-9a-zA-Z_]*

BUILTIN_IDENTIFIER_CALLS ::=
      "in"        "(" ")"
    | "fin"       "(" ")"
    | "cin"       "(" ")"
    | "draw"      "(" ")"
    | "clean_vm"  "(" ")"
    | "out"       "(" EXPRESSION ")"
    | "fout"      "(" EXPRESSION ")"
    | "cout"      "(" EXPRESSION ")"
    | "set_pixel" "(" EXPRESSION "," EXPRESSION "," EXPRESSION ")"
